### Task 1: 自动上台阶（TDD）

- [ ] 失败测试：0.4m 台阶挡路→tryStepUp 返回垫高位；1.0m 高块与 5m 墙→null；无阻挡→null。
```ts
export function tryStepUp(want: Vec3, corrected: Vec3, half: Vec3, walls: Box[], maxStep: number): Vec3 | null {
  if (Math.abs(corrected.x - want.x) < 1e-6 && Math.abs(corrected.z - want.z) < 1e-6) return null;
  for (let s = 0.05; s <= maxStep + 1e-6; s += 0.05) {
    const test = vec3(want.x, corrected.y + s, want.z);
    if (!walls.some((w) => overlaps(aabbFromCenter(test, half), w))) return test;
  }
  return null;
}
```
- [ ] playerController.update 第5步后：`if (this.grounded && this.velocityY <= 0) { const s = tryStepUp(want, corrected, this.half, this.walls, 0.55); if (s) { corrected = s; this.velocityY = 0; } }`
